import 'dart:convert';

class LocalCacheService {
  static final Map<String, String> _diskStorage = {};
  static final Map<String, DateTime> _timestampRegistry = {};

  void saveToDeviceDisk(String storageKey, List<Map<String, dynamic>> rawDataList) {
    _diskStorage[storageKey] = jsonEncode(rawDataList);
    _timestampRegistry[storageKey] = DateTime.now();
  }

  String getRawDataString(String storageKey) { return _diskStorage[storageKey] ?? "[]"; }

  void runAutomatedStorageWipeCleanup() {
    DateTime now = DateTime.now(); List<String> staleKeys = [];
    _timestampRegistry.forEach((k, t) { if (now.difference(t).inDays >= 3) staleKeys.add(k); });
    for (String k in staleKeys) { _diskStorage.remove(k); _timestampRegistry.remove(k); }
  }
}

class TimetableService {
  final List<TimetableItem> globalTimetableRepo = [];

  List<TimetableItem> fetchApprovedSchedules(String campus, String faculty, String dept, int lvl) {
    return globalTimetableRepo.where((item) => item.isApproved && item.campusCode == campus && item.faculty == faculty && item.department == dept && item.level == lvl).toList();
  }

  List<TimetableItem> fetchPendingSchedulesForReview(String campus, String faculty, String dept, int lvl) {
    return globalTimetableRepo.where((item) => !item.isApproved && item.campusCode == campus && item.faculty == faculty && item.department == dept && item.level == lvl).toList();
  }

  void submitOrVoteTimetableUpdate({required UserProfile student, required String courseCode, required String newLocation}) {
    final newItem = TimetableItem(
      id: DateTime.now().millisecondsSinceEpoch.toString(), campusCode: student.campusCode, courseCode: courseCode, lecturerName: "Staff", location: newLocation,
      timeSlot: "TBD Hour", faculty: student.faculty, department: student.department, level: student.level, isApproved: false, confirmationCount: 1,
    );
    globalTimetableRepo.add(newItem);
  }

  void castStudentValidationTick({required TimetableItem targetItem, required bool isCorrectReported}) {
    if (isCorrectReported) {
      targetItem.confirmationCount += 1;
      if (targetItem.confirmationCount >= 5) targetItem.isApproved = true; 
    }
  }
}

// lib/services/cgpa_calculator_service.dart
class CgpaCalculatorService {
  int convertGradeToPoints(String grade) {
    switch (grade.toUpperCase()) {
      case 'A': return 5; case 'B': return 4; case 'C': return 3; case 'D': return 2; case 'F': return 0; default: return 0;
    }
  }
  double calculateRsuCgpa(List<int> units, List<String> grades) {
    if (units.isEmpty || units.length != grades.length) return 0.0;
    int totalUnits = 0; int totalPoints = 0;
    for (int i = 0; i < units.length; i++) { totalUnits += units[i]; totalPoints += (units[i] * convertGradeToPoints(grades[i])); }
    return totalUnits == 0 ? 0.0 : totalPoints / totalUnits;
  }
}
